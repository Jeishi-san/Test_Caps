```php
public function toggleStar(Request $request, Document $document): JsonResponse
{
    // Authorization check
    $this->authorize('update', $document);

    $document->update(['is_starred' => !$document->is_starred]);

    return response()->json([
        'is_starred' => $document->is_starred,
        'message' => $document->is_starred ? 'Document starred' : 'Document unstarred'
    ]);
}
```

#### StarredController Creation
**File:** `app/Http/Controllers/StarredController.php` (New)
- Dedicated controller for starred items management
- `index()` method returns paginated list of user's starred documents and folders
- Combines both documents and folders in single response
- Includes permission checks for access control

```php
public function index(): JsonResponse
{
    $starredDocuments = Document::where('user_id', auth()->id())
        ->where('is_starred', true)
        ->select(['id', 'name', 'folder_id', 'created_at'])
        ->get();

    $starredFolders = Folder::where('user_id', auth()->id())
        ->where('is_starred', true)
        ->select(['id', 'name', 'parent_id', 'created_at'])
        ->get();

    return response()->json([
        'documents' => $starredDocuments,
        'folders' => $starredFolders,
    ]);
}
```

```php
public function test_user_can_star_and_unstar_document()
{
    $user = User::factory()->create();
    $document = Document::factory()->create(['user_id' => $user->id]);

    $this->actingAs($user);

    // Star the document
    $response = $this->post("/api/documents/{$document->id}/star");
    $response->assertStatus(200)
             ->assertJson(['is_starred' => true]);

    // Verify in database
    $this->assertTrue($document->fresh()->is_starred);

    // Unstar the document
    $response = $this->post("/api/documents/{$document->id}/star");
    $response->assertStatus(200)
             ->assertJson(['is_starred' => false]);
}
```
